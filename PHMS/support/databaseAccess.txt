JOIN CHECKS (COMMON)
--------------------------------------------------------------------------------
-- User + latest health entries
SELECT u.user_id, u.username, h.entry_id, h.recorded_at, h.health_score
FROM user u
LEFT JOIN health_data h ON h.user_id = u.user_id
ORDER BY h.recorded_at DESC
LIMIT 20;

-- User medications with medicine names
SELECT m.medication_id, m.user_id, med.medicine_name, m.dosage, m.frequency, m.is_critical
FROM medication m
JOIN medicine med ON med.medicine_id = m.medicine_id
ORDER BY m.medication_id DESC
LIMIT 20;

-- Medication logs with medicine names
SELECT ml.log_id, ml.user_id, med.medicine_name, ml.log_date, ml.scheduled_time, ml.status, ml.taken_at
FROM medication_log ml
JOIN medication m ON m.medication_id = ml.medication_id
JOIN medicine med ON med.medicine_id = m.medicine_id
ORDER BY ml.log_date DESC, ml.scheduled_time DESC
LIMIT 30;

-- Alerts with optional source links
SELECT a.alert_id, a.user_id, a.health_id, a.medication_log_id, a.category, a.severity, a.is_read, a.created_at
FROM alert a
ORDER BY a.created_at DESC
LIMIT 30;
