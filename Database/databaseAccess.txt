================================================================================
					Database Access & Quick Checks (SQLite)
================================================================================

PREREQS
--------------------------------------------------------------------------------
- SQLite CLI installed: sqlite3 --version
- DB file path: C:\Users\Admin\Desktop\Sem-06\2_SDP\project\PHMS\instance\phms.db

OPEN DB
--------------------------------------------------------------------------------
cd C:\Users\Admin\Desktop\Sem-06\2_SDP\project\PHMS\instance
sqlite3 phms.db

BASIC INSPECTION
--------------------------------------------------------------------------------
-- List tables
.tables

-- View schema (table-by-table)
.schema user
.schema health_data
.schema medicine
.schema medication
.schema medication_log
.schema alert

-- Peek counts
SELECT 'user' table, COUNT(*) FROM user;
SELECT 'health_data' table, COUNT(*) FROM health_data;
SELECT 'medicine' table, COUNT(*) FROM medicine;
SELECT 'medication' table, COUNT(*) FROM medication;
SELECT 'medication_log' table, COUNT(*) FROM medication_log;
SELECT 'alert' table, COUNT(*) FROM alert;

-- Sample rows
SELECT * FROM user LIMIT 5;
SELECT * FROM health_data ORDER BY recorded_at DESC LIMIT 5;
SELECT * FROM alert ORDER BY created_at DESC LIMIT 5;

INDEX & CONSTRAINT CHECKS (spot-verify)
--------------------------------------------------------------------------------
-- Unique email
PRAGMA index_list('user');

-- Medication uniqueness (user_id, medicine_id)
PRAGMA index_list('medication');
PRAGMA foreign_key_list('medication');

FOREIGN KEYS & CASCADE
--------------------------------------------------------------------------------
-- Ensure FKs are enforced (should be 1)
PRAGMA foreign_keys;
-- If 0, enable in session: PRAGMA foreign_keys = ON;

EXIT
--------------------------------------------------------------------------------
.exit
.quit

TIP: For ad-hoc data cleanup, always backup phms.db first.
